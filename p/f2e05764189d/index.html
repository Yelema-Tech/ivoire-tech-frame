Programme du service de pages ivoirTech

Les pages ivoirTech sont un site statique hébergé gratuitement sur Render, à l'adresse ivoirtech-frames.onrender.com. Un site statique ne s'endort jamais : les pages publiées restent en ligne de façon permanente.

Comment une page est publiée

  1. La page HTML est ajoutée au dépôt GitHub Yelema-Tech/ivoire-tech-frame, dans p/<identifiant>/index.html.
  2. Render remet le site en ligne avec la nouvelle page (environ une minute).
  3. Le lien est de la forme ivoirtech-frames.onrender.com/p/<identifiant>/. Modifier une page garde le même lien.

#!/usr/bin/env python3
"""Publie une page HTML autonome sur le site statique ivoirTech (ivoirtech-frames.onrender.com),
la remplace sans changer son lien, ou la supprime. Réservé au groupe Telegram ivoirTech.

Chaîne : la page est écrite dans le dépôt GitHub public Yelema-Tech/ivoire-tech-frame
(../github-ivoirtech/depot.py), puis Render redéploie le site statique (render_api.py deployer).
Les pages sont permanentes : un site statique gratuit ne s'endort pas et ne perd rien.

Usage :
  publier-page.py page.html --titre "…" [--remplacer ID]
  publier-page.py --supprimer ID
Sortie (stdout, JSON) : {"frameId": "…", "shareUrl": "…", "http": 200}
"""
import argparse, json, os, subprocess, sys, time, urllib.request, urllib.error, uuid

DOSSIER = os.path.dirname(os.path.abspath(__file__))
DEPOT = os.path.join(DOSSIER, "..", "github-ivoirtech", "depot.py")
URL = "https://ivoirtech-frames.onrender.com"
SERVICE = "srv-dbplbuhsrm7s73apsde0"  # site statique ivoirtech-frames


def echec(message):
    print(f"erreur publication ivoirTech : {message}", file=sys.stderr)
    print(f"Vérifie si l'intégration est cassée : python3 {DOSSIER}/check.py "
          f"et python3 {os.path.dirname(DEPOT)}/check.py", file=sys.stderr)
    print("Cassée → préviens l'utilisateur (règle 5). "
          "Pas cassée → dis-lui que la demande n'a pas abouti, et pourquoi.", file=sys.stderr)
    sys.exit(1)


def lancer(*args):
    r = subprocess.run([sys.executable, *args], capture_output=True, text=True)
    if r.returncode:
        echec(r.stderr.strip().splitlines()[0] if r.stderr.strip() else "échec sans message")
    return json.loads(r.stdout)


def deployer_et_attendre():
    dep = lancer(os.path.join(DOSSIER, "render_api.py"), "deployer", SERVICE)
    for _ in range(40):
        time.sleep(5)
        etat = lancer(os.path.join(DOSSIER, "render_api.py"), "etat", SERVICE)
        if etat.get("id") == dep["deployId"] and etat.get("status") == "live":
            return
        if etat.get("id") == dep["deployId"] and etat.get("status") in ("build_failed", "update_failed", "canceled"):
            echec(f"déploiement Render {etat.get('status')}")
    echec("déploiement Render toujours en cours après 200 s")


def code(lien):
    for _ in range(6):  # le CDN peut servir l'ancienne version quelques secondes
        try:
            with urllib.request.urlopen(lien, timeout=30) as r:
                return r.status
        except urllib.error.HTTPError as e:
            if e.code != 404:
                return e.code
        except Exception:
            pass
        time.sleep(5)
    return 404


def main():
    a = argparse.ArgumentParser()
    a.add_argument("page", nargs="?")
    a.add_argument("--titre", default="")
    a.add_argument("--remplacer", default="", metavar="ID")
    a.add_argument("--supprimer", default="", metavar="ID")
    o = a.parse_args()
    if o.supprimer:
        lancer(DEPOT, "supprimer", f"p/{o.supprimer}/index.html")
        deployer_et_attendre()
        print(json.dumps({"supprime": o.supprimer}))
        return
    if not o.page or not o.titre:
        a.error("page et --titre requis")
    ident = o.remplacer or uuid.uuid4().hex[:12]
    lancer(DEPOT, "ecrire", f"p/{ident}/index.html", o.page, "--message", f"page : {o.titre}")
    deployer_et_attendre()
    lien = f"{URL}/p/{ident}/"
    print(json.dumps({"frameId": ident, "shareUrl": lien, "http": code(lien)}))


if __name__ == "__main__":
    main()
