Programme du service de pages ivoirTech

Code complet du service ivoirtech-frames, hébergé sur le compte Render ivoire-tech. Un seul fichier, Node.js standard, aucune dépendance, aucune clé dedans.

  • POST /api/frames — publier une page, rend un lien partageable
  • PUT /api/frames/{id} — modifier une page sans changer son lien
  • DELETE /api/frames/{id} — supprimer une page
  • GET /share/frame/{id} — lecture publique de la page

// Service de pages ivoirTech — publie des pages HTML autonomes et rend un lien partageable.
// Même contrat que Mstudio Frames : POST/PUT/DELETE /api/frames, lecture /share/frame/{id}.
// Aucune dépendance : Node standard. Les pages vivent sur le disque persistant (DATA_DIR).
const http = require("http"), fs = require("fs"), path = require("path"), crypto = require("crypto");

const PORT = process.env.PORT || 10000;
const DATA = process.env.DATA_DIR || "/data/pages";
const KEY = process.env.FRAMES_API_KEY || "";
const BASE = (process.env.PUBLIC_URL || "").replace(/\/$/, "");
const MAX = 15 * 1024 * 1024;
fs.mkdirSync(DATA, { recursive: true });

const fichier = id => path.join(DATA, id.replace(/[^a-f0-9-]/g, "") + ".json");
const lire = id => { try { return JSON.parse(fs.readFileSync(fichier(id), "utf8")); } catch { return null; } };
const ecrire = (id, f) => { const t = fichier(id) + ".tmp"; fs.writeFileSync(t, JSON.stringify(f)); fs.renameSync(t, fichier(id)); };
const json = (res, code, obj) => { res.writeHead(code, { "Content-Type": "application/json" }); res.end(JSON.stringify(obj)); };
const autorise = req => {
  const k = req.headers["x-api-key"] || "";
  return KEY && k.length === KEY.length && crypto.timingSafeEqual(Buffer.from(k), Buffer.from(KEY));
};
const corps = req => new Promise((ok, ko) => {
  let n = 0, parts = [];
  req.on("data", c => { n += c.length; if (n > MAX) { ko(new Error("trop gros")); req.destroy(); } else parts.push(c); });
  req.on("end", () => { try { ok(JSON.parse(Buffer.concat(parts).toString() || "{}")); } catch (e) { ko(e); } });
  req.on("error", ko);
});
const lien = (req, id) => `${BASE || "https://" + req.headers.host}/share/frame/${id}`;

http.createServer(async (req, res) => {
  const url = new URL(req.url, "http://x");
  const m = url.pathname.match(/^\/api\/frames(?:\/([a-f0-9-]+))?\/?$/);
  try {
    if (req.method === "GET" && url.pathname === "/") return json(res, 200, { service: "ivoirtech-frames", ok: true });
    const s = url.pathname.match(/^\/share\/frame\/([a-f0-9-]+)\/?$/);
    if (req.method === "GET" && s) {
      const f = lire(s[1]);
      if (!f || !f.isPublic) { res.writeHead(404, { "Content-Type": "text/plain; charset=utf-8" }); return res.end("Page introuvable"); }
      res.writeHead(200, { "Content-Type": "text/html; charset=utf-8", "X-Content-Type-Options": "nosniff",
                           "Cache-Control": "no-cache", "X-Robots-Tag": "noindex" });
      return res.end(f.html);
    }
    if (m) {
      if (!autorise(req)) return json(res, 401, { error: "clé manquante ou invalide" });
      const id = m[1];
      if (req.method === "POST" && !id) {
        const b = await corps(req);
        if (typeof b.html !== "string" || !b.html) return json(res, 400, { error: "html requis" });
        const nid = crypto.randomUUID(), now = new Date().toISOString();
        ecrire(nid, { title: b.title || "", description: b.description || "", html: b.html,
                      isPublic: b.isPublic !== false, createdAt: now, updatedAt: now });
        return json(res, 201, { frameId: nid, shareUrl: lien(req, nid) });
      }
      if (id && (req.method === "GET" || req.method === "PUT" || req.method === "DELETE")) {
        const f = lire(id);
        if (!f) return json(res, 404, { error: "introuvable" });
        if (req.method === "GET") return json(res, 200, { frameId: id, shareUrl: lien(req, id), title: f.title,
                                                          isPublic: f.isPublic, createdAt: f.createdAt, updatedAt: f.updatedAt });
        if (req.method === "DELETE") { fs.unlinkSync(fichier(id)); return json(res, 200, { deleted: id }); }
        const b = await corps(req);
        for (const k of ["title", "description", "html", "isPublic"]) if (b[k] !== undefined) f[k] = b[k];
        f.updatedAt = new Date().toISOString(); ecrire(id, f);
        return json(res, 200, { frameId: id, shareUrl: lien(req, id) });
      }
    }
    json(res, 404, { error: "introuvable" });
  } catch (e) { json(res, 400, { error: String(e.message || e) }); }
}).listen(PORT, () => console.log("ivoirtech-frames prêt sur", PORT));
